class ProdGameLauncher implements GameLauncher {
    private NextBlockInfoCalculator nextBlockInfoCalculator;
    private NextDirectionCalculator nextDirectionCalculator;
    private NextBlockInfo nextBlockInfo;

    @Override
    public void launchGame(InitializedData initializedData, BaseDataLogger baseDataLogger) {
        Scanner scanner = initializedData.scannerRemainingDatas;
        StaticGameInitializedData staticGameInitializedData = initializedData.staticGameInitializedData;
        MapBuilder mapBuilder = new MapBuilder();
        nextDirectionCalculator = new NextDirectionCalculator();
        TunnelMapElementRetriever tunnelMapElementRetriever = new TunnelMapElementRetriever();
        NextBlockMapBuilder nextBlockMapBuilder = new NextBlockMapBuilder(tunnelMapElementRetriever);
        nextBlockInfoCalculator = new NextBlockInfoCalculator(tunnelMapElementRetriever, nextBlockMapBuilder);
        String[][] mapTab = mapBuilder.convertListToStringTab(staticGameInitializedData.tunnelLinesList);
        nextBlockInfo = new NextBlockInfo();
        ExitDoor exitDoor =
                new ExitDoor(
                        staticGameInitializedData.exitDoorXCord,
                        mapBuilder.mapHeight - 1
                );

        nextBlockInfo.isANonBlockingBlock = false;
        nextBlockInfo.isAFunnelCase = false;

        Coordinates heroCoordinates = new Coordinates(0, 0, "");

        int countTurn = 0;
        while (true) {
            countTurn++;

            int XI = scanner.nextInt();
            int YI = scanner.nextInt();
            String POS = scanner.next();

            heroCoordinates.cordX = XI;
            heroCoordinates.cordY = YI;
            heroCoordinates.value = mapTab[heroCoordinates.cordY][heroCoordinates.cordX];

            if(countTurn == 1){
                if(POS.equals(String.valueOf(GameDirection.RIGHT))) heroCoordinates.direction = GameDirection.RIGHT;
                else if(POS.equals(String.valueOf(GameDirection.LEFT))) heroCoordinates.direction = GameDirection.LEFT;
                else heroCoordinates.direction = GameDirection.DOWN;
            }

//            baseDataLogger.logDynamicTurnsNumber(countTurn);
//            baseDataLogger.logDynamicDatas(initializedData.hero);

            if(!nextBlockInfo.isAFunnelCase || heroCoordinates.nextCoordinates == null){
                printNextCoordinates(mapTab, heroCoordinates, exitDoor, nextBlockInfo);
            }else{
                heroCoordinates.direction = heroCoordinates.nextCoordinates.direction;
                if(heroCoordinates.nextCoordinates.direction == GameDirection.DOWN){
                    System.out.println(heroCoordinates.nextCoordinates.cordX + " " + (heroCoordinates.nextCoordinates.cordY + 1));
                }else if(heroCoordinates.nextCoordinates.direction == GameDirection.LEFT){
                    System.out.println((heroCoordinates.nextCoordinates.cordX - 1) + " " + heroCoordinates.nextCoordinates.cordY);
                }else{
                    System.out.println((heroCoordinates.nextCoordinates.cordX + 1) + " " + heroCoordinates.nextCoordinates.cordY);
                }
                heroCoordinates.nextCoordinates = null;
            }
        }
    }

    private void printNextCoordinates(String[][] mapTab, Coordinates currentCoordinates, ExitDoor exitDoor, NextBlockInfo nextBlockInfo){
        Map<GameDirection, Coordinates> nearestValues = nextDirectionCalculator.getNearestDirections(currentCoordinates, mapTab);
        List<GameDirection> nextDirections = nearestValues.keySet().stream().toList();
        for(int a = 0 ; a < nextDirections.size() ; a++){
            GameDirection nearestDirection = nextDirections.get(a);
            Coordinates nextCoordinate = nearestValues.get(nearestDirection);
            currentCoordinates.direction = nearestDirection;
            if(nextCoordinate.cordX == exitDoor.cordX && nextCoordinate.cordY == exitDoor.cordY) System.out.println(nextCoordinate.cordX + " " + nextCoordinate.cordY);
            else {
                this.nextBlockInfo = nextBlockInfoCalculator.isNextBlockNonBlocking(mapTab, currentCoordinates, nextCoordinate, nextBlockInfo);
                if(nextBlockInfo.isANonBlockingBlock){
                    System.out.println(nextCoordinate.cordX + " " + nextCoordinate.cordY);
                    return;
                }
            }
        }
    }
}


class TestGameLauncher implements GameLauncher {
    @Override
    public void launchGame(InitializedData initializedData, BaseDataLogger baseDataLogger) {
        Game game = new Game();

        baseDataLogger.logStaticDatas(initializedData.staticGameInitializedData);

        Hero hero = initializedData.hero;
        ExitDoor exitDoor = initializedData.exitDoor;

        int countTurn = 0;
        while (heroNotOnExitDoor(hero, exitDoor)){

            countTurn++;
            baseDataLogger.logDynamicTurnsNumber(countTurn);
            baseDataLogger.logDynamicDatas(initializedData.hero);

            game.play();
        }
    }

    private boolean heroNotOnExitDoor(Hero hero, ExitDoor exitDoor){
        return !heroOnExitDoor(hero, exitDoor);
    }

    private boolean heroOnExitDoor(Hero hero, ExitDoor exitDoor){
        return hero.cordX == exitDoor.cordX && hero.cordY == exitDoor.cordY;
    }
}

class ProdDataInitializer implements DataInitializer {
    @Override
    public InitializedData initData() {
        Scanner scanner = new Scanner(System.in);
        int mapWidth = scanner.nextInt(); // number of columns.
        int mapHeight = scanner.nextInt(); // number of rows.

        if (scanner.hasNextLine()) {
            scanner.nextLine();
        }

        List<String> tunnelLinesList = new ArrayList<>();

        for (int i = 0; i < mapHeight; i++) {
            String LINE = scanner.nextLine(); // represents a line in the grid and contains W integers. Each integer represents one room of a given type.
            tunnelLinesList.add(LINE);
        }
        int exitDoorXCord = scanner.nextInt(); // the coordinate along the X axis of the exit (not useful for this first mission, but must be read).

        InitializedData initializedData = new InitializedData();
        initializedData.scannerRemainingDatas = scanner;
        initializedData.staticGameInitializedData =
                new StaticGameInitializedData(tunnelLinesList, mapWidth, mapHeight, exitDoorXCord);

        return initializedData;
    }
}

        return tunnelElementList.stream()
                .filter(tunnelElementConversion -> tunnelElementConversion.mapValue.equals(value))
                .findFirst()
                .get()
                .tunnelElement;
    }

        return tunnelElementList.stream()

                .get()

        return tunnelElementList.stream()
                .filter(tunnelElementConversion -> tunnelElementConversion.tunnelElement == tunnelElement)
                .findFirst()
                .get()
                .mapValue;
    }

class MapBuilder {
    private static String DELIMITER = " ";
    public int mapWidth;
    public int mapHeight;
    public String[][] convertListToStringTab(List<String> map){
        String[][] stringTab = initStringTab(map);
        for(int a = 0 ; a < stringTab.length ; a++){
            String[] stringLineTab = map.get(a).split(DELIMITER);
            for(int b = 0 ; b < stringTab[0].length ; b++){
                stringTab[a][b] = stringLineTab[b];
            }
        }
        return stringTab;
    }

    private String[][] initStringTab(List<String> map){
        int mapWidth = map.get(0).split(DELIMITER).length;
        return initStringTab(mapWidth, map.size());
    }

    private String[][] initStringTab(int width, int height){
        this.mapWidth = width;
        this.mapHeight = height;
        return new String[height][width];
    }
}

class NextBlockInfo {
    public boolean isANonBlockingBlock;
    public boolean isAFunnelCase;
}


class NextBlockInfoCalculator {
    NextBlockMapBuilder nextBlockMapBuilder;
    TunnelMapElementRetriever tunnelMapElementRetriever;
    public NextBlockInfoCalculator(TunnelMapElementRetriever tunnelMapElementRetriever, NextBlockMapBuilder nextBlockMapBuilder){
        this.tunnelMapElementRetriever = tunnelMapElementRetriever;
        this.nextBlockMapBuilder = nextBlockMapBuilder;
    }
    public NextBlockInfo isNextBlockNonBlocking(String[][] mapTab, Coordinates currentCoordinates, Coordinates nextCoordinates, NextBlockInfo nextBlockInfo){
        nextBlockInfo.isANonBlockingBlock = false;
        nextBlockInfo.isAFunnelCase = false;
        TunnelElement startingTunnelElement = tunnelMapElementRetriever.getByStringValue(currentCoordinates.value);
        TunnelElement nextTunnelElement = tunnelMapElementRetriever.getByStringValue(nextCoordinates.value);
        if(nextTunnelElement == TunnelElement.DIRECTION_BLOCK) return nextBlockInfo;
        else if(nextTunnelElement == TunnelElement.DIRECTION_TWO_CURVES_REVERSE){
            if(currentCoordinates.direction == GameDirection.DOWN){
                nextCoordinates.direction = GameDirection.RIGHT;
                return funnelCase(mapTab, currentCoordinates, nextCoordinates, nextBlockInfo, nextTunnelElement);
            }else if(currentCoordinates.direction == GameDirection.RIGHT){
                nextCoordinates.direction = GameDirection.DOWN;
                return funnelCase(mapTab, currentCoordinates, nextCoordinates, nextBlockInfo, nextTunnelElement);
            }else return nextBlockInfo;
        }
        else if(nextTunnelElement == TunnelElement.DIRECTION_TWO_CURVES){
            if(currentCoordinates.direction == GameDirection.DOWN){
                nextCoordinates.direction = GameDirection.LEFT;
                return funnelCase(mapTab, currentCoordinates, nextCoordinates, nextBlockInfo, nextTunnelElement);
            }else if(currentCoordinates.direction == GameDirection.LEFT){
                nextCoordinates.direction = GameDirection.DOWN;
                return funnelCase(mapTab, currentCoordinates, nextCoordinates, nextBlockInfo, nextTunnelElement);
            }else return nextBlockInfo;
        }
        else {
            nextBlockInfo.isANonBlockingBlock = nextBlockMapBuilder.buildMap().get(startingTunnelElement).get(currentCoordinates.direction).contains(nextTunnelElement);
            return nextBlockInfo;
        }
    }

    private NextBlockInfo funnelCase(String[][] mapTab, Coordinates currentCoordinates, Coordinates nextCoordinates, NextBlockInfo nextBlockInfo, TunnelElement nextTunnelElement){
        nextBlockInfo.isANonBlockingBlock = getInfo(nextCoordinates, mapTab, nextTunnelElement);
        if(nextBlockInfo.isANonBlockingBlock) {
            currentCoordinates.nextCoordinates = nextCoordinates;
            nextBlockInfo.isAFunnelCase = true;
        }
        return nextBlockInfo;
    }

    private boolean getInfo(Coordinates nextCoordinates, String[][] mapTab, TunnelElement nextTunnelElement){
        try {
            String nextNextCordValue = getMapValue(mapTab, nextCoordinates);
            TunnelElement nextNextTunnelElement = tunnelMapElementRetriever.getByStringValue(nextNextCordValue);
            return nextBlockMapBuilder.buildMap().get(nextTunnelElement).get(nextCoordinates.direction).contains(nextNextTunnelElement);
        }catch (ArrayIndexOutOfBoundsException e){
            return false;
        }
    }



    private String getMapValue(String[][] mapTab, Coordinates nextCoordinates){
        if(nextCoordinates.direction == GameDirection.DOWN){
            return mapTab[nextCoordinates.cordY + 1][nextCoordinates.cordX];
        }else if(nextCoordinates.direction == GameDirection.RIGHT){
            return mapTab[nextCoordinates.cordY][nextCoordinates.cordX + 1];
        }else{
            return mapTab[nextCoordinates.cordY][nextCoordinates.cordX - 1];
        }
    }
}



class NearestMapConstructor {
    public void constructNearestMap(GameDirection direction, int cordX, int cordY, Map<GameDirection, Coordinates> nearestValues, String[][] stringTab){
        try {
            nearestValues.put(direction, new Coordinates(cordX, cordY, stringTab[cordY][cordX]));
        }catch (ArrayIndexOutOfBoundsException e){}
    }
}



class NextDirectionCalculator{
    private NearestMapConstructor nearestMapConstructor;
    public NextDirectionCalculator(){
        this.nearestMapConstructor = new NearestMapConstructor();
    }
    public Map<GameDirection, Set<GameDirection>> directionsRegardingPrevious =
            Map.of(
                    GameDirection.RIGHT,
                    Set.of(
                            GameDirection.RIGHT,
                            GameDirection.DOWN
                    ),
                    GameDirection.LEFT,
                    Set.of(
                            GameDirection.LEFT,
                            GameDirection.DOWN
                    ),
                    GameDirection.DOWN,
                    Set.of(
                            GameDirection.RIGHT,
                            GameDirection.DOWN,
                            GameDirection.LEFT
                    )
            );

    public Map<GameDirection, Coordinates> getNearestDirections(Coordinates coordinates, String[][] stringTab){
        Map<GameDirection, Coordinates> nearestValues = new HashMap<>();
        if(coordinates.direction == null){
            nearestMapConstructor.constructNearestMap(
                    GameDirection.RIGHT,
                    coordinates.cordX + 1,
                    coordinates.cordY,
                    nearestValues,
                    stringTab
            );
            nearestMapConstructor.constructNearestMap(
                    GameDirection.LEFT,
                    coordinates.cordX - 1,
                    coordinates.cordY,
                    nearestValues,
                    stringTab
            );
            nearestMapConstructor.constructNearestMap(
                    GameDirection.DOWN,
                    coordinates.cordX,
                    coordinates.cordY + 1,
                    nearestValues,
                    stringTab
            );
        }else{
            directionsRegardingPrevious.get(coordinates.direction).stream()
                    .forEach(direction -> {
                        if(direction == GameDirection.RIGHT) {
                            nearestMapConstructor.constructNearestMap(
                                    GameDirection.RIGHT,
                                    coordinates.cordX + 1,
                                    coordinates.cordY,
                                    nearestValues,
                                    stringTab
                            );
                        }
                        else if(direction == GameDirection.LEFT){
                            nearestMapConstructor.constructNearestMap(
                                    GameDirection.LEFT,
                                    coordinates.cordX - 1,
                                    coordinates.cordY,
                                    nearestValues,
                                    stringTab
                            );
                        }else{
                            nearestMapConstructor.constructNearestMap(
                                    GameDirection.DOWN,
                                    coordinates.cordX,
                                    coordinates.cordY + 1,
                                    nearestValues,
                                    stringTab
                            );
                        }
                    });
        }
        return nearestValues;
    }
}

    public TunnelElementException(TunnelElement tunnelElement) {
        super("TunnelElement: " + tunnelElement + " does not exists");
    }
}

class GameMapUpdater {
    int mapWidth;
    int mapHeight;
    private static String DELIMITER = " ";
    TunnelMapElementRetriever tunnelMapElementRetriever;

    public void updateExitDoorValues(GameElementsContainer gameElementsContainer, Coordinates coordinates){
        gameElementsContainer.exitDoor.previousValueRegistered = true;
        gameElementsContainer.exitDoor.previousCordX = gameElementsContainer.exitDoor.cordX;
        gameElementsContainer.exitDoor.previousCordY = gameElementsContainer.exitDoor.cordY;
        gameElementsContainer.exitDoor.cordX = coordinates.cordX;
        gameElementsContainer.exitDoor.cordY = coordinates.cordY;
    }
    public void updateGameMap(GameElementsContainer gameElementsContainer, TunnelElement tunnelElementToUpdate){
        GameMap gameMap = gameElementsContainer.gameMap;
        Hero hero = gameElementsContainer.hero;
        ExitDoor exitDoor = gameElementsContainer.exitDoor;
        this.tunnelMapElementRetriever = new TunnelMapElementRetriever();
        MapBuilder mapBuilder = new MapBuilder();

        String[][] stringTab = mapBuilder.convertListToStringTab(gameMap.map);
        mapWidth = mapBuilder.mapWidth;
        mapHeight = mapBuilder.mapHeight;

        if(tunnelElementToUpdate == TunnelElement.HERO) restorePreviousElement(stringTab, hero);
        else restorePreviousElement(stringTab, exitDoor);

        putElementInMap(exitDoor, stringTab);
        stringTab = putElementInMap(hero, stringTab);
        gameMap.map = convertMapCharTabToList(stringTab);
    }

    private List<String> convertMapCharTabToList(String[][] gameMapString){
        return Arrays.stream(gameMapString)
                .map(element -> Arrays.stream(element).collect(Collectors.joining(DELIMITER)))
                .toList();
    }

    private void restorePreviousElement(String[][] gameMapChar, BaseElement baseElement){
        secureMapIntegration(baseElement.previousCordX, baseElement.previousCordY);
        if(baseElement.previousValueRegistered){
            gameMapChar[baseElement.previousCordY][baseElement.previousCordX] = baseElement.previousValue;
        }
    }
    private String[][] putElementInMap(BaseElement baseElement, String[][] gameMapChar){
        secureMapIntegration(baseElement.cordX, baseElement.cordY);
        baseElement.previousCordX = baseElement.cordX;
        baseElement.previousCordY = baseElement.cordY;
        baseElement.previousValue = gameMapChar[baseElement.cordY][baseElement.cordX];
        baseElement.previousValueRegistered = true;
        String elementValue;
        if(baseElement instanceof Hero){
             elementValue = getTunnelElementMapValue(TunnelElement.HERO);
        }else{
            elementValue = getTunnelElementMapValue(TunnelElement.EXIT);
        }
        gameMapChar[baseElement.cordY][baseElement.cordX] = elementValue;
        return gameMapChar;
    }

    private String getTunnelElementMapValue(TunnelElement tunnelElement){
        return tunnelMapElementRetriever.getByTunnelElementValue(tunnelElement);
    }

    private void secureMapIntegration(int cordX, int cordY){
        if(cordX > mapWidth) throw new ElementOutOfMapWidth(cordX, mapWidth);
        if((cordY + 1) > mapHeight) throw new ElementOutOfMapHeight(cordY, mapHeight);
    }
}

class TunnelDirectionLogPrinter {
    public void printLog(TunnelElement tunnelElement){
        System.out.println(tunnelElement + " :");
        List<String> listToPrint =
            TunnelElementLog.tunnelElementLogsMap
                .get(tunnelElement);
        listToPrint.stream()
                .forEach(System.out::println);
        System.out.println();
    }

    public void printLog(int intValuevalue){
        String stringValue = "" + intValuevalue;
        printLog(stringValue);
    }

    public void printLog(String value){
        TunnelMapElementRetriever tunnelMapElementRetriever = new TunnelMapElementRetriever();
        System.out.print(value + " : ");
        TunnelElement tunnelElement =
                tunnelMapElementRetriever.getByMapValue(TunnelElementGameMapValue.valueOf(value));
        printLog(tunnelElement);
    }
}
